  <script>
    const taskInput = document.getElementById('taskInput');
    const addBtn = document.getElementById('addBtn');
    const clearDoneBtn = document.getElementById('clearDoneBtn');
    const taskList = document.getElementById('taskList');
    const emptyState = document.getElementById('emptyState');

    const statTotal = document.getElementById('statTotal');
    const statDone = document.getElementById('statDone');
    const tagFilter = document.getElementById('tagFilter');

    const filters = document.querySelectorAll('.filterBtn');
    let currentFilter = 'all';

    let tasks = [];
    // Estructura: { id, title, done, createdAt }

    function uid(){
      return Math.random().toString(16).slice(2) + Date.now().toString(16);
    }

    function fmtDate(ts){
      const d = new Date(ts);
      return d.toLocaleDateString(undefined, { year:'numeric', month:'short', day:'2-digit' });
    }

    function applyFilter(list){
      if(currentFilter === 'all') return list;
      if(currentFilter === 'todo') return list.filter(t => !t.done);
      if(currentFilter === 'done') return list.filter(t => t.done);
      return list;
    }

    function render(){
      taskList.innerHTML = '';
      const total = tasks.length;
      const doneCount = tasks.filter(t => t.done).length;

      statTotal.textContent = total;
      statDone.textContent = doneCount;

      const view = applyFilter(tasks);
      emptyState.style.display = view.length ? 'none' : 'block';

      tagFilter.textContent = 'Filtro: ' + (currentFilter==='all'?'Todo':currentFilter==='todo'?'Pendientes':'Completadas');

      view.sort((a,b)=>b.createdAt - a.createdAt);

      view.forEach(t => {
        const el = document.createElement('div');
        el.className = 'task' + (t.done ? ' done' : '');

        el.innerHTML = `
          <div class="left">
            <button class="check" aria-label="Marcar completada" title="Completar/Descompletar" style="border:0;cursor:pointer;background:transparent;color:inherit;padding:0;">
              <svg width="14" height="14" viewBox="0 0 20 20" fill="none" xmlns="http://www.w3.org/2000/svg">
                <path d="M16.7 5.9L8.4 14.2L3.3 9.1" stroke="rgba(52,211,153,1)" stroke-width="2.4" stroke-linecap="round" stroke-linejoin="round"/>
              </svg>
            </button>
            <div style="min-width:0;">
              <div class="title">${escapeHtml(t.title)}</div>
              <div class="meta">
                <span class="tag">${t.done ? '✅ Hecha' : '⏳ Pendiente'}</span>
                <span>📅 ${fmtDate(t.createdAt)}</span>
              </div>
            </div>
          </div>

          <div class="actions">
            <button class="btnDanger" title="Eliminar">Eliminar</button>
          </div>
        `;

        const checkBtn = el.querySelector('.check');
        const delBtn = el.querySelector('.btnDanger');

        checkBtn.addEventListener('click', () => {
          tasks = tasks.map(x => x.id === t.id ? { ...x, done: !x.done } : x);
          render();
        });

        delBtn.addEventListener('click', () => {
          tasks = tasks.filter(x => x.id !== t.id);
          render();
        });

        taskList.appendChild(el);
      });
    }

    function escapeHtml(str){
      return str.replace(/[&<>"']/g, m => ({
        '&':'&amp;',
        '<':'&lt;',
        '>':'&gt;',
        '"':'&quot;',
        "'":'&#039;'
      }[m]));
    }

    function addTask(){
      const title = taskInput.value.trim();
      if(!title) return;

      tasks.unshift({
        id: uid(),
        title,
        done: false,
        createdAt: Date.now()
      });

      taskInput.value = '';
      render();
      taskInput.focus();
    }

    addBtn.addEventListener('click', addTask);
    taskInput.addEventListener('keydown', (e) => {
      if(e.key === 'Enter') addTask();
    });

    clearDoneBtn.addEventListener('click', () => {
      tasks = tasks.filter(t => !t.done);
      render();
    });

    filters.forEach(btn => {
      btn.addEventListener('click', () => {
        currentFilter = btn.dataset.filter;
        filters.forEach(b => b.classList.remove('active'));
        btn.classList.add('active');
        render();
      });
    });

    // inicial
    render();
  </script>
